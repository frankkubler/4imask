# pyproject.toml porte la version affichee par l'application : sans --add-data,
# le bundle affiche "inconnue" alors que le lancement depuis les sources est correct.
# Sous Windows, remplacer les ":" des --add-data par des ";".
pyinstaller 4imask_anonymizer.py --noconsole --add-data "res:res" --add-data "pyproject.toml:." --icon="res/icons/4imask.ico" --hidden-import=ultralytics --collect-submodules=openvino --collect-binaries=openvino --collect-data=openvino --hidden-import=skimage._shared.geometry --splash="res/icons/splash.png"

# Commande courante. S'ajoutent deux documents ouverts depuis l'interface : docs/ pour
# l'information RGPD, et LICENSE pour le lien AGPL, jusqu'ici absent du bundle.
pyinstaller 4imask_anonymizer.py --noconsole --add-data "res:res" --add-data "docs:docs" --add-data "LICENSE:." --add-data "pyproject.toml:." --icon="res/icons/4imask.ico" --hidden-import=ultralytics --collect-submodules=openvino --collect-binaries=openvino --collect-data=openvino --hidden-import=skimage._shared.geometry --splash="res/icons/splash.png"
